cmake_minimum_required(VERSION 3.20)
project(fxp_physics VERSION 0.1.0 LANGUAGES CXX)
set(FXP_PHYSICS_TOP_LEVEL OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(FXP_PHYSICS_TOP_LEVEL ON)
endif()
option(FXP_PHYSICS_BUILD_TESTS "Build headless physics tests" ${FXP_PHYSICS_TOP_LEVEL})
option(FXP_PHYSICS_BUILD_EXAMPLES "Build headless example" ${FXP_PHYSICS_TOP_LEVEL})
option(FXP_PHYSICS_BUILD_BENCHMARKS "Build reproducible headless benchmarks" ${FXP_PHYSICS_TOP_LEVEL})
option(FXP_PHYSICS_SANITIZERS "Enable address/undefined sanitizers" OFF)
if(NOT TARGET fxp::fxp)
  add_subdirectory(external/fxp_math)
endif()
find_package(Threads REQUIRED)
add_library(fxp_physics src/geometry.cpp src/world.cpp src/contacts.cpp src/solver.cpp src/serialization.cpp)
add_library(fxp::physics ALIAS fxp_physics)
target_include_directories(fxp_physics PUBLIC include)
target_link_libraries(fxp_physics PUBLIC fxp::fxp Threads::Threads)
target_compile_features(fxp_physics PUBLIC cxx_std_17)
set_target_properties(fxp_physics PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(fxp_physics PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(fxp_physics PRIVATE -Wall -Wextra -Wpedantic)
  if(FXP_PHYSICS_SANITIZERS)
    target_compile_options(fxp_physics PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(fxp_physics PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(FXP_PHYSICS_BUILD_TESTS)
  enable_testing()
  add_executable(fxp_physics_tests tests/tests.cpp)
  target_link_libraries(fxp_physics_tests PRIVATE fxp::physics)
  target_compile_definitions(fxp_physics_tests PRIVATE FXP_PHYSICS_TEST_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
  add_test(NAME physics COMMAND fxp_physics_tests)
  add_test(NAME determinism COMMAND fxp_physics_tests --reference)
  add_executable(fxp_physics_geometry_tests tests/geometry_tests.cpp)
  target_link_libraries(fxp_physics_geometry_tests PRIVATE fxp::physics)
  add_test(NAME geometry_cache COMMAND fxp_physics_geometry_tests)
  add_executable(fxp_physics_contact_cache_tests tests/contact_cache_tests.cpp)
  target_link_libraries(fxp_physics_contact_cache_tests PRIVATE fxp::physics)
  add_test(NAME contact_cache COMMAND fxp_physics_contact_cache_tests)
  add_executable(fxp_physics_solver_tests tests/solver_tests.cpp)
  target_link_libraries(fxp_physics_solver_tests PRIVATE fxp::physics)
  add_test(NAME solver COMMAND fxp_physics_solver_tests)
endif()
if(FXP_PHYSICS_BUILD_EXAMPLES)
  add_executable(fxp_physics_example examples/headless.cpp)
  target_link_libraries(fxp_physics_example PRIVATE fxp::physics)
endif()
if(FXP_PHYSICS_BUILD_BENCHMARKS OR FXP_PHYSICS_BUILD_TESTS)
  add_executable(fxp_physics_benchmark benchmarks/benchmark.cpp)
  target_link_libraries(fxp_physics_benchmark PRIVATE fxp::physics)
  if(MSVC)
    target_compile_options(fxp_physics_benchmark PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(fxp_physics_benchmark PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(FXP_PHYSICS_BUILD_TESTS)
    add_test(NAME playground_determinism COMMAND fxp_physics_benchmark
      --scene playground --ticks 240
      --check-reference "${CMAKE_CURRENT_SOURCE_DIR}/tests/playground_reference.csv")
    set_tests_properties(playground_determinism PROPERTIES TIMEOUT 600)
    foreach(scene kinematic dynamic stack)
      add_test(NAME benchmark_smoke_${scene} COMMAND fxp_physics_benchmark
        --scene ${scene} --bodies 32 --ticks 3 --warmup 1 --samples 2 --queries 3)
    endforeach()
  endif()
endif()
